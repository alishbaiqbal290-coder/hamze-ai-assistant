let isListening = false; let recognition; function speakUrdu(text){ try{ speechSynthesis.cancel(); let u = new SpeechSynthesisUtterance(text); u.lang = "ur-PK"; speechSynthesis.speak(u); }catch{} } function startVoice(){ const SR = window.SpeechRecognition || window.webkitSpeechRecognition; if(!SR){ alert("Chrome استعمال کریں"); return; } if(isListening){ isListening = false; try{ recognition.stop(); }catch{} document.getElementById("micBtn").classList.remove("listening"); document.getElementById("status").innerText = "مائیک بند ہے"; return; } recognition = new SR(); recognition.lang = "ur-PK"; recognition.onstart = function(){ isListening = true; document.getElementById("micBtn").classList.add("listening"); document.getElementById("status").innerText = "سن رہا ہوں... آن ہے"; }; recognition.onend = function(){ document.getElementById("micBtn").classList.remove("listening"); isListening = false; }; recognition.onresult = function(e){ let cmd = e.results[0][0].transcript; document.getElementById("command").value = cmd; document.getElementById("response").innerText = "آپ نے کہا: " + cmd; speakUrdu("حکم مل گیا " + cmd); if(cmd.includes("یوٹیوب")) window.open("https://www.youtube.com","_blank"); if(cmd.includes("واٹس ایپ")) window.open("https://wa.me/","_blank"); isListening = false; }; recognition.start(); }
